# Python dependencies for the capture and build tools: everything under
# tools/kit/, the per-package copies under systems/*/tools/, and the repo-level
# tools that read or write images. `make setup` installs this list into .venv,
# so an activated .venv gives the tools a `python3` that can see them; on any
# other host, `pip install -r tools/requirements-capture.txt` into the
# interpreter `PYTHON` (or `python3`) resolves to. Docs: docs/CAPTURE.md.
#
# Binaries are not listed: ffmpeg (mirror_media.py --compress-videos),
# tesseract (redact_images.py OCR) and sips (macOS thumbnails) live on PATH.
#
# Pinned: a reference package's rebuild must change nothing (AGENTS.md), and
# quantised PNGs and measured text widths depend on the Pillow, numpy and
# fonttools versions. Pillow, numpy and markdown match the owner's .venv;
# fonttools is the release current when pinned. Bump them deliberately and
# rebuild the reference packages in the same change.

pillow==13.0.0       # quantize.py, crops.py, srgb.py, shot.py, store.py,
            # kit/*.mjs inline `python3 -c "from PIL import ..."` quantize calls,
            # make_icons.py, redact_images.py, catalog.py gallery thumbnails,
            # linear build_guide.py/build_library.py, polaris build_vectors.py
numpy==2.5.4         # redact_images.py
markdown==3.11.1     # site.py, linear build_guide.py
fonttools[woff]==4.68.0  # polaris build_vectors.py (measures text widths; the woff
                 # extra pulls brotli, needed to decode inter.woff2 to TTF)
